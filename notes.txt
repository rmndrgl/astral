Future<List<Map<String, dynamic>>> getRecentDrafts(int limit) async {
  final db = await sqlt.getDB();
  return await db.query(
    'drafts',
    orderBy: 'created_at DESC',
    limit: limit,
  );
}

Future<Map<String, dynamic>?> getDraftByUuid(String uuid) async {
  final db = await sqlt.getDB();
  final result = await db.query(
    'drafts',
    where: 'uuid = ?',
    whereArgs: [uuid],
  );
  return result.isNotEmpty ? result.first : null;
}

Future<List<Map<String, dynamic>>> getDrafts() async {
  final db = await sqlt.getDB();
  return await db.query('drafts', orderBy: 'created_at DESC');
}
